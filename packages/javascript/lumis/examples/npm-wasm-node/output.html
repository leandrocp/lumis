npm-wasm-node + Lumis

This example runs in Node and imports @lumis-sh/wasm-elixir directly.

defmodule Lumis.Release do
  def manifest(env) do
    %{target: env, profile: :release, artifact: "lumis-#{env}.beam"}
  end

  def print(env) do
    env
    |> manifest()
    |> IO.inspect(label: "release")
  end
end

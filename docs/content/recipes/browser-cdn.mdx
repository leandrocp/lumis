---
title: Browsers / CDN
description: Highlight code directly in Browsers with ESM CDN imports.
---



```html
<script type="module">
  import {highlight} from 'https://esm.sh/@lumis-sh/lumis'
  import {
    htmlInline,
  } from 'https://esm.sh/@lumis-sh/lumis/formatters'
  import javascript from 'https://esm.sh/@lumis-sh/lumis/langs/javascript'
  import frappe from 'https://esm.sh/@lumis-sh/themes/catppuccin_frappe'

  const output = document.getElementById('output')
  output.innerHTML = await highlight(
    'const x = 1',
    htmlInline({language: javascript, theme: frappe})
  )
</script>
<div id="output"></div>
```

See [Where parsers come from](/usage/javascript#where-parsers-come-from) for how
a browser fetches and caches parsers, and
[Custom resolvers](/usage/javascript#custom-resolvers) to serve them yourself.
